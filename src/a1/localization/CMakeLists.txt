cmake_minimum_required(VERSION 3.0.2)
project(a1_localization)
add_compile_options(-std=c++14 -Wall -Wextra -Wpedantic)
find_package(catkin REQUIRED COMPONENTS diagnostic_msgs diagnostic_updater pcl_conversions pcl_ros roscpp rosgraph_msgs sensor_msgs std_srvs geometry_msgs nav_msgs tf2 tf2_geometry_msgs tf2_ros)
find_package(OpenSSL REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io)
catkin_package(CATKIN_DEPENDS diagnostic_msgs diagnostic_updater pcl_conversions pcl_ros roscpp rosgraph_msgs sensor_msgs std_srvs geometry_msgs nav_msgs tf2 tf2_geometry_msgs tf2_ros)
include_directories(${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_executable(pointcloud_adapter src/pointcloud_adapter.cpp)
target_link_libraries(pointcloud_adapter ${catkin_LIBRARIES})
add_executable(localization_pose_adapter src/localization_pose_adapter.cpp)
target_link_libraries(localization_pose_adapter ${catkin_LIBRARIES})
add_executable(localization_map_manager src/localization_map_manager.cpp)
target_link_libraries(localization_map_manager ${catkin_LIBRARIES} ${PCL_LIBRARIES} OpenSSL::Crypto)
install(TARGETS pointcloud_adapter localization_pose_adapter localization_map_manager RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
catkin_install_python(PROGRAMS scripts/localization_preflight_check.py scripts/localization_stability_gate.py scripts/localization_supervisor.py
  scripts/localization_validation_recorder.py
  DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  find_package(rostest REQUIRED)
  catkin_install_python(PROGRAMS test/localization_pose_adapter_runtime_test.py
    test/localization_supervisor_runtime_test.py test/localization_map_manager_runtime_test.py test/supervisor_test_estimator.py
    DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
  catkin_add_nosetests(test/localization_contract_test.py)
  add_rostest(test/localization_pose_adapter.test)
  add_rostest(test/localization_supervisor.test)
  add_rostest(test/localization_map_manager.test)
endif()
